cmake_minimum_required(VERSION 3.10)
project(pak_tests C)

enable_testing()

option(PAK_WASM "Test the WAMR bindings" ON)
set(PAK_WASM_TEST_MODULE "" CACHE FILEPATH "Path to the WASM dummy module")
set(PAK_WASM_BT_TEST_MODULE "" CACHE FILEPATH "Path to the WASM Bluetooth test module")
set(PAK_WAMR_THREAD_TEST_MODULE "" CACHE FILEPATH "Path to the shared-memory WAMR threading probe")
set(PAK_WASM_WIFI_TEST_MODULE "" CACHE FILEPATH "Path to the WASM Wi-Fi test module")
set(PAK_WASM_SOCKET_TEST_MODULE "" CACHE FILEPATH "Path to the WASM socket test module")
set(PAK_WASM_USB_TEST_MODULE "" CACHE FILEPATH "Path to the WASM USB test module")
set(PAK_WASM_VCAM_TEST_MODULE "" CACHE FILEPATH "Path to the WASM vcam test module")
set(PAK_WASM_VCAM_TCP_TEST_MODULE "" CACHE FILEPATH "Path to the WASM vcam TCP test module")
set(PAK_WASM_FUJI_SETUP_TEST_MODULE "" CACHE FILEPATH "Path to the WASM libfuji setup test module")
set(PAK_VCAM_LIB "" CACHE FILEPATH "Path to vcam's libusb-vcam shared library")
set(PAK_VCAM_EXECUTABLE "" CACHE FILEPATH "Path to the vcam executable")

set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
add_library(pak_test SHARED
	${ROOT}/src/common.c ${ROOT}/src/gc.c ${ROOT}/src/sandbox.c
	${ROOT}/src/cli_runtime.c fake_bluetooth.c fake_wifi.c)
target_include_directories(pak_test PUBLIC ${ROOT}/include)
target_compile_definitions(pak_test PRIVATE PAK_FAKE_BLUETOOTH=1)

if(PAK_WASM)
	find_package(PkgConfig REQUIRED)
	pkg_check_modules(LIBUSB REQUIRED IMPORTED_TARGET libusb-1.0)
	find_library(WAMR_LIB NAMES iwasm REQUIRED)
	include(CheckFunctionExists)
	set(CMAKE_REQUIRED_LIBRARIES ${WAMR_LIB} m pthread ${CMAKE_DL_LIBS})
	unset(PAK_HAS_WAMR_SOCKET_DUP CACHE)
	check_function_exists(wasm_runtime_dup_wasi_socket PAK_HAS_WAMR_SOCKET_DUP)
	unset(CMAKE_REQUIRED_LIBRARIES)
	if(NOT PAK_HAS_WAMR_SOCKET_DUP)
		message(FATAL_ERROR "WAMR must include the socket duplication patch in patches/wamr-2.4.5-dup-wasi-socket.patch")
	endif()
	target_sources(pak_test PRIVATE
		${ROOT}/src/wasm/runtime.c ${ROOT}/src/wasm/printf.c
		${ROOT}/src/wasm/pointer_table.c ${ROOT}/src/wasm/libusb.c
		${ROOT}/src/wasm/bluetooth.c ${ROOT}/src/wasm/wifi.c ${ROOT}/src/wasm/socket.c)
	target_link_libraries(pak_test PRIVATE ${WAMR_LIB} PkgConfig::LIBUSB m)
endif()

add_executable(sandbox_test sandbox_test.c ${ROOT}/src/sandbox.c)
target_compile_options(sandbox_test PRIVATE -UNDEBUG)
add_test(NAME sandbox COMMAND sandbox_test)

function(pak_test name source)
	add_executable(${name}_test ${source})
	target_compile_options(${name}_test PRIVATE -UNDEBUG)
	target_link_libraries(${name}_test PRIVATE pak_test)
endfunction()

pak_test(fake_bluetooth fake_bluetooth_test.c)
add_test(NAME fake_bluetooth COMMAND fake_bluetooth_test)
pak_test(fake_wifi fake_wifi_test.c)
add_test(NAME fake_wifi COMMAND fake_wifi_test)

if(PAK_WASM)
	if(PAK_WAMR_THREAD_TEST_MODULE)
		find_package(Threads REQUIRED)
		add_executable(wamr_thread_test wamr_thread_test.c)
		target_include_directories(wamr_thread_test PRIVATE ${ROOT}/wamr/core/iwasm/include)
		target_link_libraries(wamr_thread_test PRIVATE ${WAMR_LIB} Threads::Threads m ${CMAKE_DL_LIBS})
		add_test(NAME wamr_thread COMMAND wamr_thread_test "${PAK_WAMR_THREAD_TEST_MODULE}")
		set_tests_properties(wamr_thread PROPERTIES TIMEOUT 10)
	endif()
	if(PAK_WASM_BT_TEST_MODULE)
		pak_test(wasm_bluetooth wasm_bluetooth_test.c)
		add_test(NAME wasm_bluetooth COMMAND wasm_bluetooth_test "${PAK_WASM_BT_TEST_MODULE}")
	endif()
	if(PAK_WASM_WIFI_TEST_MODULE)
		pak_test(wasm_wifi wasm_wifi_test.c)
		add_test(NAME wasm_wifi COMMAND wasm_wifi_test "${PAK_WASM_WIFI_TEST_MODULE}")
	endif()
	if(PAK_WASM_TEST_MODULE)
		pak_test(wasm_dummy wasm_dummy.c)
		add_test(NAME wasm_dummy COMMAND wasm_dummy_test "${PAK_WASM_TEST_MODULE}")
	endif()
	if(PAK_WASM_SOCKET_TEST_MODULE)
		pak_test(wasm_socket wasm_socket_test.c)
		add_test(NAME wasm_socket COMMAND wasm_socket_test "${PAK_WASM_SOCKET_TEST_MODULE}")
	endif()
	if(PAK_WASM_USB_TEST_MODULE)
		add_library(wasm_usb_mock SHARED wasm_usb_mock.c)
		target_link_libraries(wasm_usb_mock PRIVATE PkgConfig::LIBUSB)
		pak_test(wasm_usb wasm_socket_test.c)
		add_test(NAME wasm_usb COMMAND ${CMAKE_COMMAND} -E env "LD_PRELOAD=$<TARGET_FILE:wasm_usb_mock>" $<TARGET_FILE:wasm_usb_test> "${PAK_WASM_USB_TEST_MODULE}")
	endif()
	if(PAK_WASM_VCAM_TEST_MODULE AND PAK_VCAM_LIB)
		pak_test(wasm_vcam wasm_socket_test.c)
		add_test(NAME wasm_vcam COMMAND ${CMAKE_COMMAND} -E env "LD_PRELOAD=${PAK_VCAM_LIB}" "VCAM_USB_MODEL=fuji_x_h1" $<TARGET_FILE:wasm_vcam_test> "${PAK_WASM_VCAM_TEST_MODULE}")
	endif()
	if(PAK_WASM_VCAM_TCP_TEST_MODULE AND PAK_VCAM_EXECUTABLE)
		pak_test(wasm_vcam_tcp wasm_vcam_tcp_test.c)
		add_test(NAME wasm_vcam_tcp COMMAND wasm_vcam_tcp_test "${PAK_VCAM_EXECUTABLE}" "${PAK_WASM_VCAM_TCP_TEST_MODULE}")
		set_tests_properties(wasm_vcam_tcp PROPERTIES TIMEOUT 25 RESOURCE_LOCK vcam_tcp_port)
	endif()
	if(PAK_WASM_FUJI_SETUP_TEST_MODULE AND PAK_VCAM_EXECUTABLE)
		pak_test(wasm_fuji_setup wasm_vcam_tcp_test.c)
		add_test(NAME wasm_fuji_setup COMMAND wasm_fuji_setup_test "${PAK_VCAM_EXECUTABLE}" "${PAK_WASM_FUJI_SETUP_TEST_MODULE}")
		set_tests_properties(wasm_fuji_setup PROPERTIES TIMEOUT 60 RESOURCE_LOCK vcam_tcp_port)
	endif()
endif()
